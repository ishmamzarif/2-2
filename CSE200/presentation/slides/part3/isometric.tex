% Shared fixed camera and upright labels for every computation graph.
\definecolor{isoInk}{HTML}{263452}
\definecolor{isoPaper}{HTML}{F3F4F9}
\tikzset{graphviewport/.style={},
  baseroad/.style={draw=rblue!60,line width=1.25pt},
  activeroad/.style={line width=2.1pt},
  routeroad/.style={activeroad,-{Stealth[length=1.8mm]},shorten >=4.5mm,shorten <=4.5mm},
  roadweight/.style={font=\fontsize{9}{10}\selectfont,text=isoInk,fill=isoPaper,
    inner sep=1.6pt,rounded corners=1pt}}

\newcommand{\GraphShiftX}{-.20}
% Affine camera: these same coefficients also project the grid and its L path.
\newcommand{\IsoPoint}[3]{\coordinate (#1) at
  ({.5+\GraphShiftX+1.08*(.39*(#2)+.10*(#3))},{2.8+1.08*(-.65-.065*(#2)+.29*(#3))});}
\newcommand{\IsoCoordinates}{%
  \foreach \n/\x/\y in \OptimalityCoordinates {\IsoPoint{\n}{\x}{\y}}
  \IsoPoint{manhattanTurn}{\CoordSX}{\CoordNY}}
% Shared uninterrupted four-corner plane and shallow shadow.
\newcommand{\IsoPlatePath}{(plateA)--(plateB)--(plateC)--(plateD)--cycle}
\newcommand{\IsoPlatform}{%
  \foreach \n/\x/\y in {A/-.05/1.94,B/6.95/.9,C/7.7/3.61,D/.75/4.65}{%
    \coordinate (plate\n) at ({1.08*\x-.04+\GraphShiftX},{1.08*\y-.224});
    \coordinate (shadow\n) at ($(plate\n)+(0,-.15)$);}
  \fill[isoInk!12] (shadowA)--(shadowB)--(shadowC)--(shadowD)--cycle;
  \draw[draw=rblue!14,fill=isoPaper,line width=.6pt] \IsoPlatePath;}
\newcommand{\IsoDisc}[3]{%
  \begingroup\colorlet{discStroke}{#3}
  \fill[isoInk,opacity=.10] ($(#1)+(.04,-.08)$) ellipse (.40 and .28);
  \fill[discStroke!23] ($(#1)+(-.39,0)$)--++(0,.16)
    arc[start angle=180,end angle=360,x radius=.39,y radius=.34]--++(0,-.16)
    arc[start angle=0,end angle=-180,x radius=.39,y radius=.34];
  \draw[draw=discStroke,fill=#2,fill opacity=1,line width=.85pt]
    ($(#1)+(0,.16)$) ellipse (.39 and .34);
  \endgroup}
\newcommand{\IsoRing}[2]{\draw[#2,line width=.75pt] ($(#1)+(0,.16)$) ellipse (.455 and .4);}
\newcommand{\NodeCostContent}[3]{%
  \node[font=\sffamily\fontsize{10}{11}\selectfont\bfseries,text=isoInk,inner sep=0pt]
    at ($(#1)+(0,.30)$) {#2};
  \node[font=\fontsize{9}{10}\selectfont,text=cbOrange,inner sep=0pt]
    at ($(#1)+(0,0)$) {$#3$};}
\newcommand{\IsoSimpleDisc}[4]{%
  \IsoDisc{#1}{#3}{#4}
  \node[font=\sffamily\fontsize{10}{11}\selectfont\bfseries,text=isoInk,inner sep=0pt]
    at ($(#1)+(0,.16)$) {#2};}
\newcommand{\RoadEdge}[3]{\draw[#3] (#1)--(#2);}
\newcommand{\IsoGrid}{%
  \begin{scope}\clip \IsoPlatePath;
  \foreach \x in {0,...,16}{%
    \IsoPoint{gridStart}{\x}{0}\IsoPoint{gridEnd}{\x}{8}
    \pgfmathtruncatemacro{\major}{mod(\x,4)}
    \ifnum\major=0 \draw[black!19,line width=.4pt] (gridStart)--(gridEnd);
    \else\draw[black!11,line width=.3pt] (gridStart)--(gridEnd);\fi}
  \foreach \y in {0,...,8}{%
    \IsoPoint{gridStart}{0}{\y}\IsoPoint{gridEnd}{16}{\y}
    \pgfmathtruncatemacro{\major}{mod(\y,4)}
    \ifnum\major=0 \draw[black!19,line width=.4pt] (gridStart)--(gridEnd);
    \else\draw[black!11,line width=.3pt] (gridStart)--(gridEnd);\fi}
  \end{scope}}
\newcommand{\IsoCitySurface}{%
  \IsoCoordinates\IsoPlatform
  \ifnum\IsoGridVisible=1 \IsoGrid\fi
  \foreach \a/\b/\w in \StoryEdges {\RoadEdge{\a}{\b}{baseroad}}
  \foreach \a/\b in \IsoParents {\RoadEdge{\a}{\b}{draw=rblue!72,line width=1.5pt}}
  \foreach \a/\b in \IsoActive {\RoadEdge{\a}{\b}{draw=cbOrange,activeroad}}
  \foreach \a/\b in \IsoRoute {\ifnum\IsoReverse=1
    \RoadEdge{\b}{\a}{draw=\IsoRouteColor,routeroad}
    \else\RoadEdge{\a}{\b}{draw=\IsoRouteColor,routeroad}\fi}
  \foreach \a/\b/\w in \StoryEdges {%
    \coordinate (iw) at ($(\a)!.5!(\b)$);
    \IfSame{\a/\b}{N/B}{\coordinate (iw) at ($(iw)+(0,-.13)$);}{}
    \IfSame{\a/\b}{D/G}{\coordinate (iw) at ($(iw)+(-.32,0)$);}{}
    \node[roadweight] at (iw) {\w};}
  \ifnum\IsoLVisible>0
    \draw[cbOrange,dashed,line width=1.5pt,-{Stealth[length=1.8mm]}] (N)--(manhattanTurn);
    \node[font=\small,text=cbOrange,fill=isoPaper,inner sep=1pt]
      at ($(N)!.70!(manhattanTurn)+(0,.75)$) {$\Delta x=\ManhattanDX$};
    \ifnum\IsoLVisible<3
    \draw[cbOrange,dashed,line width=1.5pt,-{Stealth[length=1.8mm]}] (manhattanTurn)--(S);
    \draw[cbOrange!55,line width=.5pt] ($(manhattanTurn)!.5!(S)+(.14,0)$)
      --($(S)+(.52,-.62)$);
    \node[font=\small,text=cbOrange,fill=isoPaper,inner sep=1pt,anchor=north east]
      at ($(S)+(.55,-.70)$) {$\Delta y=\ManhattanDY$};\fi
  \fi
  \foreach \n in \StoryNodes {%
    \def\podFill{white}\def\podStroke{rblue!70}
    \IfMember{\n}{\IsoOpen}{\def\podFill{rbluelt}}{}
    \IfMember{\n}{\IsoClosed}{\def\podFill{closedc}\def\podStroke{neutralg}}{}
    \IfSame{\n}{\IsoCurrent}{\def\podFill{hlyellow}\def\podStroke{cbOrange}}{}
    \IfMember{\n}{\IsoRouteNodes}{\def\podFill{\IsoRouteColor!12}\def\podStroke{\IsoRouteColor}}{}
    \IsoDisc{\n}{\podFill}{\podStroke}
    \IfMember{\n}{\IsoFocus}{\IsoRing{\n}{cbOrange}}{}
  }
  \IsoRing{D}{cbGreen}\IsoRing{S}{cbRed}}
