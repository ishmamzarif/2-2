\documentclass{article}
\usepackage{tikz}
\usetikzlibrary{positioning}

\tikzset{
    block/.style={
        draw,
        rounded corners,
        minimum width=2.6cm,
        minimum height=1cm,
        align=center,
        fill=blue!10
    },
    neuron/.style={
        circle,
        draw,
        minimum size=8mm
    },
    line/.style={
        ->,
        thick,
        red
    }
}
